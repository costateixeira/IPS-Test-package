  <description
               value="Test plan for the **IPS Server** actor of [HL7 International Patient Summary (IPS) 2.0.1](https://hl7.org/fhir/uv/ips/).
An IPS Server is a FHIR server acting as an IPS Creator by providing conformant
IPS documents in response to FHIR API requests (CapabilityStatement ips-server).
Each `suite` below is a `Rule:` of the Gherkin feature named by `suite.input.file`, and each `suite.test` one of its `Scenario:` lines, matched by the scenario's identifier. The assertions are the scenario's `Then` steps, verbatim.
The tests declare which specification obligations they cover (9 element obligations across the plan); the coverage matrix is checked mechanically against the specification package."/>

           value="To declare, in a machine-readable and runnable form, which behaviours a system claiming conformance to the IPS Server actor must demonstrate."/>

    <reference
               value="http://hl7.org/fhir/uv/ips/CapabilityStatement/ips-server"/>

                 value="The IPS Server actor — a FHIR server acting as an IPS Creator through the $summary operation — is the system under test."/>
  </scope>
  <runner value="https://www.itb.ec.europa.eu/docs/guides/latest/"/>
  <suite>
    <name value="The server declares the IPS capabilities"/>
    <description
                 value="Rule: The server declares the IPS capabilities — the scenarios grouped under it in the feature file."/>
    <input>
      <name value="gherkin-script"/>
      <file value="ips-server.feature"/>
    </input>
    <test>
      <name
            value="ips-server-001 The CapabilityStatement declares the $summary operation"/>
      <operation value="gherkin/Scenario"/>
      <assertion>
        <severity value="error"/>

      </assertion>
      <assertion>
        <severity value="error"/>
        <human
               value="$capability at &quot;CapabilityStatement.rest.where(mode='server').resource.where(type='Patient').operation.where(name='summary').exists()&quot; should be true"/>
      </assertion>
      <assertion>
        <severity value="error"/>
        <human
               value="$capability at &quot;CapabilityStatement.instantiates.where($this.startsWith('http://hl7.org/fhir/uv/ips/CapabilityStatement/ips-server')).exists() or CapabilityStatement.rest.resource.where(type='Patient').operation.where(name='summary').definition.exists()&quot; should be true"/>
      </assertion>
    </test>
  </suite>
  <suite>
    <name value="$summary returns an IPS document"/>
    <description
                 value="Rule: $summary returns an IPS document — the scenarios grouped under it in the feature file."/>
    <input>
      <name value="gherkin-script"/>
      <file value="ips-server.feature"/>
    </input>
    <test>
      <name
            value="ips-server-002 $summary by patient identifier returns a Bundle (IPS) with identifier, timestamp and the composition first"/>
      <description
                   value="Covers Bundle-uv-ips: Bundle.identifier, Bundle.timestamp, Bundle.entry:composition."/>
      <operation value="gherkin/Scenario"/>
      <assertion>
        <severity value="error"/>
        <human value="$response.status should be 200"/>
      </assertion>
      <assertion>
        <severity value="error"/>
        <human
               value="$ips should conform to &quot;http://hl7.org/fhir/uv/ips/StructureDefinition/Bundle-uv-ips&quot; ignoring slicing errors"/>

      <assertion>
        <severity value="error"/>
        <human
               value="$ips at &quot;Bundle.type&quot; should be &quot;document&quot;"/>
      </assertion>
      <assertion>
        <severity value="error"/>
        <human
               value="$ips at &quot;Bundle.identifier.exists()&quot; should be true"/>
      </assertion>
      <assertion>
        <severity value="error"/>
        <human
               value="$ips at &quot;Bundle.timestamp.exists()&quot; should be true"/>
      </assertion>
      <assertion>
        <severity value="error"/>
        <human
               value="$ips at &quot;Bundle.entry.first().resource.ofType(Composition).exists()&quot; should be true"/>

      <assertion>

      </assertion>
      <assertion>
        <severity value="error"/>

        <severity value="error"/>
